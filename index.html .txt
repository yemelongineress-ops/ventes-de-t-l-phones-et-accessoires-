<script>
let panier = [];
let produits = [
{nom:"iPhone 13", prix:350000, cat:"iphone", img:"https://fdn2.gsmarena.com/vv/pics/apple/apple-iphone-13-01.jpg", badge:"🔥 Promo"},
{nom:"Samsung S21", prix:300000, cat:"samsung", img:"https://fdn2.gsmarena.com/vv/pics/samsung/samsung-galaxy-s21-5g-1.jpg", badge:"Nouveau"},
{nom:"Batterie iPhone", prix:15000, cat:"accessoire", img:"https://fdn2.gsmarena.com/vv/pics/apple/apple-iphone-13-01.jpg"}
];

function afficherProduits(liste){
    let html="";
    liste.forEach(p=>{
        html+=`
        <div class="produit">
            ${p.badge ? `<div class="badge">${p.badge}</div>`:""}
            <img src="${p.img}">
            <h4>${p.nom}</h4>
            <p>${p.prix} FCFA</p>
            <button onclick="add('${p.nom}',${p.prix})">Ajouter</button>
        </div>`;
    });
    document.getElementById("produits").innerHTML = html;
}

function add(nom, prix){
    panier.push({nom, prix});
    document.getElementById("count").innerText = panier.length;
}

function filtrer(cat){
    if(cat==="all") afficherProduits(produits);
    else afficherProduits(produits.filter(p=>p.cat===cat));
}

function rechercher(){
    let val = document.getElementById("search").value.toLowerCase();
    afficherProduits(produits.filter(p=>p.nom.toLowerCase().includes(val)));
}

function commander(){
    let msg="Commande:%0A";
    panier.forEach(p=>{
        msg+=p.nom+" "+p.prix+" FCFA%0A";
    });
    window.open("https://wa.me/237659330336?text="+msg);
}

window.onload = ()=> afficherProduits(produits);
